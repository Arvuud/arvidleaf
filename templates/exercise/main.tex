\begin{exercise}[4 points]
  Show that $\sum_{k=1}^{n} k = \frac{n(n+1)}{2}$.
\end{exercise}

\begin{solution}
  By induction on $n$.
\end{solution}

\begin{exercise}
  Compute
  \begin{enumerate}[label=(\alph*)]
    \item $\int_0^\pi \sin x \,\mathrm{d}x$,
    \item $\lim_{x\to 0} \frac{\sin x}{x}$.
  \end{enumerate}
\end{exercise}
